library(DESeq2)
library(pheatmap)
library(ggplot2)

# Load data
counts <- read.csv("counts.csv", row.names=1, check.names=FALSE)
design <- read.csv("design.csv", row.names=1)

design$group <- factor(design$group, levels = c("A", "B"))  # Ensuring correct factor levels

# Remove non-sample columns
counts <- counts[, colnames(counts) %in% rownames(design)]

dds <- DESeqDataSetFromMatrix(countData = counts,
                              colData = design,
                              design = ~ group)  # Adjust based on your experiment

# Run DESeq2 pipeline
dds <- DESeq(dds)

# Get results with adjusted p-values
res <- results(dds, alpha=0.05)

# Order results by adjusted p-value
resOrdered <- res[order(res$padj), ]

# Save results to file
write.csv(as.data.frame(resOrdered), file="DEG_results.csv")

# Identify significant DEGs
sigGenes <- rownames(resOrdered[which(resOrdered$padj < 0.05), ])

# PCA plot
rld <- rlog(dds, blind=FALSE)
pcaData <- plotPCA(rld, intgroup="group", returnData=TRUE)
ggplot(pcaData, aes(PC1, PC2, color=group)) +
  geom_point(size=3) +
  theme_minimal() +
  labs(title="PCA Plot of Samples", x="PC1", y="PC2")

# Heatmap of significant DEGs
pheatmap(assay(rld)[sigGenes, ], scale="row", cluster_rows=TRUE, cluster_cols=TRUE,
         annotation_col=design, show_rownames=FALSE, main="Heatmap of Significant DEGs")

# Summary of results
num_sig_genes <- sum(resOrdered$padj < 0.05, na.rm=TRUE)
cat("Number of significant DEGs: ", num_sig_genes, "\n")
